// Simplified illustration — the actual production code is proprietary.
// This mirrors the logic driving the simulator above.

function recalculateDealSize(Booking $booking, string $reason): void
{
    $previous = $booking->deal_size;
    $updated  = $this->calculator->compute($booking);

    // idempotent — nothing to log if nothing actually changed
    if ($previous === $updated) {
        return;
    }

    DB::transaction(function () use ($booking, $previous, $updated, $reason) {
        $booking->update(['deal_size' => $updated]);

        DealSizeAudit::create([
            'booking_id'     => $booking->id,
            'previous_value' => $previous,
            'new_value'      => $updated,
            'reason'         => $reason,        // e.g. 'refund_applied', 'gst_updated'
            'triggered_by'   => auth()->id() ?? 'system',
        ]);
    });
}
