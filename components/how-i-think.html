// Simplified illustration — the actual production code is proprietary.
// This mirrors the stage-rule engine driving the simulator above.

final class StageRuleEngine
{
    public function decide(array $summary, Booking $booking): ?StageMove
    {
        // rules read the AI summary, not the raw audio
        if ($this->mentions($summary, 'refund')) {
            return new StageMove('refund_pending', 'refund_request');
        }
        if ($summary['intent'] === 'confirm') {
            return new StageMove('confirmed', 'trip_confirmed');
        }
        if ($summary['sentiment'] === 'escalated') {
            return new StageMove('escalated', 'priority_escalation');
        }

        return null; // nothing matched → nothing changes, nothing logged
    }
}
